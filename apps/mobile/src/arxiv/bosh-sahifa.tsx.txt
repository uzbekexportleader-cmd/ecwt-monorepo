import React from 'react';
import { Pressable, RefreshControl, ScrollView, StyleSheet, View } from 'react-native';
import { Text } from '../../src/components/AppText';
import { Ionicons } from '@expo/vector-icons';
import { useRouter } from 'expo-router';
import { useQueryClient } from '@tanstack/react-query';

import {
  useApplications,
  useCompletion,
  useMarketplaces,
  useSubsidies,
  useUnreadCount,
} from '../../src/api/queries';
import { useAuthStore } from '../../src/store/auth';
import { EcwtApiError } from '../../src/api/client';
import {
  Button,
  Card,
  Chip,
  ErrorView,
  LoadingView,
  ProgressBar,
  Screen,
  SectionHeader,
} from '../../src/components/ui';
import { ApplicationCard, SubsidyCard } from '../../src/components/domain';
import { colors, layout, radius, spacing, typography } from '../../src/theme';
import { useT } from '../../src/i18n';

export default function HomeScreen() {
  const t = useT();
  const router = useRouter();
  const qc = useQueryClient();
  const user = useAuthStore((s) => s.user);

  const completion = useCompletion();
  const subsidies = useSubsidies();
  const applications = useApplications();
  const marketplaces = useMarketplaces();
  const unread = useUnreadCount();

  const loading = completion.isLoading || subsidies.isLoading;
  const hasError = completion.isError || subsidies.isError;
  const refreshing = completion.isFetching && !completion.isLoading;

  const activeApps = (applications.data ?? []).filter(
    (a) => !['PAID', 'REJECTED', 'CANCELLED'].includes(a.status),
  );
  const topSubsidies = (subsidies.data ?? [])
    .filter((s) => s.eligibility.verdict !== 'NOT_ELIGIBLE')
    .slice(0, 3);

  const firstName = user?.fullName?.split(' ')[1] ?? user?.fullName?.split(' ')[0];

  return (
    <Screen
      refreshControl={
        <RefreshControl
          refreshing={refreshing}
          onRefresh={() => void qc.invalidateQueries()}
          tintColor={colors.primary}
        />
      }
    >
      {/* Sarlavha */}
      <View style={[layout.rowBetween, { marginBottom: spacing.xl }]}>
        <View style={{ flex: 1 }}>
          <Text style={typography.caption}>ECWT</Text>
          <Text style={typography.h1} numberOfLines={1}>
            {firstName ? t('home.greeting', { name: firstName }) : t('home.greetingAnon')}
          </Text>
        </View>
        <Pressable onPress={() => router.push('/notifications')} hitSlop={10} style={styles.bell}>
          <Ionicons name="notifications-outline" size={22} color={colors.text} />
          {(unread.data?.count ?? 0) > 0 ? <View style={styles.bellDot} /> : null}
        </Pressable>
      </View>

      {loading ? <LoadingView label="Ma’lumotlar yuklanmoqda..." /> : null}

      {hasError ? (
        <Card style={{ borderColor: colors.danger }}>
          <ErrorView
            message={
              completion.error instanceof EcwtApiError && completion.error.isNetworkError
                ? 'Serverga ulanib bo‘lmadi. Telefoningiz va kompyuter bitta Wi-Fi tarmog‘ida ekanini tekshiring.'
                : 'Ma’lumotlarni yuklab bo‘lmadi.'
            }
            onRetry={() => void qc.refetchQueries()}
          />
        </Card>
      ) : null}

      {/* Profil progressi */}
      {completion.data ? (
        <Card>
          <View style={[layout.rowBetween, { marginBottom: spacing.md }]}>
            <Text style={typography.bodyStrong}>
              {t('home.profileProgress', { percent: completion.data.percent })}
            </Text>
            <Text style={[typography.caption, { color: colors.primary }]}>
              {completion.data.items.filter((i) => i.status === 'DONE').length}/
              {completion.data.items.length}
            </Text>
          </View>
          <ProgressBar percent={completion.data.percent} />

          <View style={{ marginTop: spacing.lg, gap: spacing.sm }}>
            {completion.data.items
              .filter((i) => i.status !== 'DONE')
              .slice(0, 3)
              .map((item) => (
                <Pressable
                  key={item.key}
                  onPress={() => item.route && router.push(item.route as never)}
                  style={layout.row}
                >
                  <Ionicons
                    name={item.status === 'WARN' ? 'alert-circle' : 'close-circle'}
                    size={16}
                    color={item.status === 'WARN' ? colors.warning : colors.danger}
                  />
                  <Text style={[typography.small, { marginLeft: spacing.sm, flex: 1 }]}>
                    {item.label}
                  </Text>
                  <Ionicons name="chevron-forward" size={14} color={colors.textMuted} />
                </Pressable>
              ))}
          </View>

          {completion.data.percent < 100 ? (
            <View style={{ marginTop: spacing.lg }}>
              <Button
                title={t('home.completeProfile')}
                variant="secondary"
                onPress={() => router.push('/(tabs)/profile')}
              />
            </View>
          ) : null}
        </Card>
      ) : null}

      {/* Tezkor amallar */}
      <SectionHeader title={t('home.quickActions')} />
      <View style={styles.quickGrid}>
        <QuickAction
          icon="ribbon-outline"
          label={t('services.subsidy')}
          onPress={() => router.push('/(tabs)/opportunities')}
        />
        <QuickAction
          icon="add-circle-outline"
          label={t('product.add')}
          onPress={() => router.push('/products/new')}
        />
        <QuickAction
          icon="document-text-outline"
          label={t('application.title')}
          onPress={() => router.push('/(tabs)/applications')}
        />
        <QuickAction
          icon="folder-open-outline"
          label={t('documents.title')}
          onPress={() => router.push('/profile/documents')}
        />
      </View>

      {/* Imkoniyatlar */}
      <SectionHeader
        title={t('home.myOpportunities')}
        actionLabel={t('home.seeAll')}
        onAction={() => router.push('/(tabs)/opportunities')}
      />
      {topSubsidies.length ? (
        topSubsidies.map((s) => (
          <SubsidyCard key={s.id} subsidy={s} onPress={() => router.push(`/subsidy/${s.id}`)} />
        ))
      ) : (
        <Card>
          <Text style={typography.small}>{t('home.noOpportunities')}</Text>
        </Card>
      )}

      {/* Faol arizalar */}
      <SectionHeader
        title={t('home.activeApplications')}
        actionLabel={activeApps.length ? t('home.seeAll') : undefined}
        onAction={() => router.push('/(tabs)/applications')}
      />
      {activeApps.length ? (
        activeApps
          .slice(0, 2)
          .map((a) => (
            <ApplicationCard
              key={a.id}
              application={a}
              onPress={() => router.push(`/applications/${a.id}`)}
            />
          ))
      ) : (
        <Card>
          <Text style={typography.small}>{t('home.noApplications')}</Text>
        </Card>
      )}

      {/* ECWT xizmatlari */}
      <SectionHeader title={t('home.services')} />
      <Card onPress={() => router.push('/services')}>
        <View style={[layout.row, { gap: spacing.md }]}>
          <View style={[styles.serviceIcon, { backgroundColor: colors.accentSoft }]}>
            <Ionicons name="globe-outline" size={22} color={colors.accent} />
          </View>
          <View style={{ flex: 1 }}>
            <Text style={typography.bodyStrong}>{t('onboarding.3.title')}</Text>
            <Text style={typography.caption}>{t('services.exportDesc')}</Text>
          </View>
          <Ionicons name="chevron-forward" size={18} color={colors.textMuted} />
        </View>
      </Card>

      {/* Marketplace'lar */}
      <SectionHeader title={t('home.marketplaces')} />
      <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ gap: spacing.sm }}>
        {(marketplaces.data ?? []).map((m) => (
          <View key={m.id} style={styles.mpChip}>
            <Text style={{ fontSize: 20 }}>{m.logoEmoji ?? '🛒'}</Text>
            <Text style={typography.caption} numberOfLines={1}>
              {m.name}
            </Text>
            {m.isMock ? <Chip label="demo" tone="gold" /> : null}
          </View>
        ))}
      </ScrollView>

      {/* AI yordamchi */}
      <View style={{ marginTop: spacing['2xl'] }}>
        <Button
          title={t('ai.title')}
          icon="chatbubble-ellipses-outline"
          variant="secondary"
          onPress={() => router.push('/assistant')}
        />
      </View>
    </Screen>
  );
}

function QuickAction({
  icon,
  label,
  onPress,
}: {
  icon: keyof typeof Ionicons.glyphMap;
  label: string;
  onPress: () => void;
}) {
  return (
    <Pressable onPress={onPress} style={({ pressed }) => [styles.quickItem, pressed && { opacity: 0.8 }]}>
      <View style={styles.quickIcon}>
        <Ionicons name={icon} size={22} color={colors.primary} />
      </View>
      <Text style={[typography.caption, { color: colors.text, textAlign: 'center' }]} numberOfLines={2}>
        {label}
      </Text>
    </Pressable>
  );
}

const styles = StyleSheet.create({
  bell: {
    width: 44,
    height: 44,
    borderRadius: radius.md,
    backgroundColor: colors.surface,
    borderWidth: 1,
    borderColor: colors.border,
    alignItems: 'center',
    justifyContent: 'center',
  },
  bellDot: {
    position: 'absolute',
    top: 10,
    right: 12,
    width: 8,
    height: 8,
    borderRadius: 4,
    backgroundColor: colors.danger,
  },
  quickGrid: { flexDirection: 'row', flexWrap: 'wrap', gap: spacing.md },
  quickItem: {
    width: '47%',
    backgroundColor: colors.surface,
    borderWidth: 1,
    borderColor: colors.border,
    borderRadius: radius.lg,
    padding: spacing.lg,
    alignItems: 'center',
    gap: spacing.sm,
  },
  quickIcon: {
    width: 44,
    height: 44,
    borderRadius: radius.md,
    backgroundColor: colors.primarySoft,
    alignItems: 'center',
    justifyContent: 'center',
  },
  serviceIcon: {
    width: 44,
    height: 44,
    borderRadius: radius.md,
    alignItems: 'center',
    justifyContent: 'center',
  },
  mpChip: {
    width: 96,
    padding: spacing.md,
    borderRadius: radius.md,
    backgroundColor: colors.surface,
    borderWidth: 1,
    borderColor: colors.border,
    alignItems: 'center',
    gap: 4,
  },
});
